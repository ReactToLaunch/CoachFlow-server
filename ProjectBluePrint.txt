[ ] Rule 5: Run Lint/Prettier before every commit.

--------------------------------------------------------------------------------
2. DATABASE SCHEMAS (THE "DATA CONTRACT")
--------------------------------------------------------------------------------

A. AUTHENTICATION MODULE
Model Name: User
Collection: users
Fields:
  - name: String (Required)
  - phone: String (Unique, Index, Required) -> This is the Login ID
  - password: String (Hashed)
  - role: String (Enum: 'admin', 'student')
  - isActive: Boolean (Default: true)
  - fcmToken: String (For Push Notifications)
  - createdAt: Date
  - updatedAt: Date

B. STUDENT DATA MODULE
Model Name: StudentProfile
Collection: student_profiles
Fields:
  - user: ObjectId (Ref: User) -> Link to Auth
  - enrollmentId: String (Unique, e.g., "SOL-2026-001")
  - batch: ObjectId (Ref: Batch)
  - parentPhone: String (Required)
  - photoUrl: String (Default: "")
  - address: String
  - gender: String (Enum: 'Male', 'Female')
  - joiningDate: Date
  - feeStatus: String (Enum: 'PAID', 'DUE', 'OVERDUE')
  - nextFeeDueDate: Date

C. BATCH MODULE
Model Name: Batch
Collection: batches
Fields:
  - name: String (e.g., "JEE Class 12 - A")
  - code: String (Unique, e.g., "JEE-12A")
  - subjects: Array of Strings ["Physics", "Maths"]
  - schedule: String (Description, e.g., "Mon-Fri 4pm")
  - isActive: Boolean

D. ATTENDANCE MODULE
Model Name: Attendance
Collection: attendance
Fields:
  - date: Date (Index) -> Store as Midnight UTC
  - batch: ObjectId (Ref: Batch)
  - totalStudents: Number
  - presentCount: Number
  - absentStudents: Array of ObjectIds (Ref: User) -> List of WHO was absent
  - markedBy: ObjectId (Ref: User) -> The Admin who marked it

E. ACADEMIC RESULTS MODULE
Model Name: Result
Collection: results
Fields:
  - testName: String (e.g., "Weekly Physics Test 4")
  - testDate: Date
  - batch: ObjectId (Ref: Batch)
  - totalMarks: Number (e.g., 300)
  - photoUrl: String (Optional, image of physical result sheet)
  - records: Array of Objects
      [
        {
          student: ObjectId (Ref: User),
          marks: Number,
          rank: Number
        }
      ]

F. COMMUNICATION MODULE
Model Name: Notice
Collection: notices
Fields:
  - title: String
  - content: String (Rich Text/HTML)
  - type: String (Enum: 'URGENT', 'INFO', 'RESULT', 'HOLIDAY')
  - targetBatches: Array of ObjectIds (Ref: Batch) -> [] means All Batches
  - attachments: Array of Strings (Image URLs)
  - postedBy: ObjectId (Ref: User)
  - createdAt: Date
